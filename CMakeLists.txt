cmake_minimum_required(VERSION 3.24)
project(xgen_memory VERSION 0.1.0 DESCRIPTION "Bounded explicit C11 memory services" LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(XGM_ABI_VERSION 1)
set(XGM_COMPONENTS "allocator;pool;size_class;tracking;libc_allocator" CACHE STRING "Explicit memory components")
set(XGM_STATUS_SOURCE_DIR "" CACHE PATH "Explicit xgen-status source checkout")
option(XGM_BUILD_TESTS "Build host GoogleTest tests" OFF)
option(XGM_ENABLE_COVERAGE "Instrument production C objects" OFF)
if(XGM_BUILD_TESTS AND CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Host tests must be built separately from firmware")
endif()
if(XGM_ENABLE_COVERAGE AND (NOT XGM_BUILD_TESTS OR CMAKE_CROSSCOMPILING OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU"))
    message(FATAL_ERROR "Coverage requires a GCC host test build")
endif()
set(_components ${XGM_COMPONENTS})
foreach(component IN LISTS XGM_COMPONENTS)
    if(NOT component MATCHES "^(allocator|pool|size_class|tracking|libc_allocator)$")
        message(FATAL_ERROR "Unknown memory component: ${component}")
    endif()
    if(component STREQUAL "size_class")
        list(APPEND _components pool)
    endif()
    if(NOT component STREQUAL "allocator")
        list(APPEND _components allocator)
    endif()
endforeach()
list(REMOVE_DUPLICATES _components)
if("pool" IN_LIST _components OR "tracking" IN_LIST _components)
    if(NOT TARGET xgs::status)
        if(XGM_STATUS_SOURCE_DIR)
            add_subdirectory("${XGM_STATUS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dependencies/status")
        else()
            find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED COMPONENTS status)
        endif()
    endif()
endif()
function(xgm_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow $<$<COMPILE_LANGUAGE:C>:-Wstrict-prototypes>)
    endif()
endfunction()
foreach(component IN ITEMS allocator pool size_class tracking libc_allocator)
    if(NOT component IN_LIST _components)
        continue()
    endif()
    set(source ${component})
    if(component MATCHES "^(size_class|tracking)$")
        set(source ${component}_allocator)
    endif()
    add_library(xgm_${component} STATIC src/${source}.c)
    add_library(xgm::${component} ALIAS xgm_${component})
    set_target_properties(xgm_${component} PROPERTIES EXPORT_NAME ${component}
        C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
        XGM_VERSION ${PROJECT_VERSION} XGM_ABI_VERSION ${XGM_ABI_VERSION}
        EXPORT_PROPERTIES "XGM_VERSION;XGM_ABI_VERSION")
    target_compile_features(xgm_${component} PUBLIC c_std_11)
    target_include_directories(xgm_${component} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    if(NOT component STREQUAL "allocator")
        target_link_libraries(xgm_${component} PUBLIC xgm::allocator)
    endif()
    if(component MATCHES "^(pool|tracking)$")
        target_link_libraries(xgm_${component} PUBLIC xgs::status)
    elseif(component STREQUAL "size_class")
        target_link_libraries(xgm_${component} PUBLIC xgm::pool)
    endif()
    xgm_warnings(xgm_${component})
    if(XGM_ENABLE_COVERAGE)
        target_compile_options(xgm_${component} PRIVATE --coverage -O0 -g)
        target_link_options(xgm_${component} INTERFACE --coverage)
    endif()
endforeach()
if(XGM_BUILD_TESTS)
    enable_language(CXX)
    find_package(GTest 1.16.0 EXACT CONFIG REQUIRED)
    enable_testing()
    add_subdirectory(tests)
endif()
